%------------------------------------------------------------------ two decisions
\begin{frame}{Two decisions that behave completely differently}
\vspace{0.8em}
\begin{columns}[T]
\column{0.47\textwidth}
\textbf{Loading is easy.}\\[0.5em]
{\small For a \emph{fixed} order, the cheapest policy is known in closed form:
\emph{Keep Tool Needed Soonest} (Tang \& Denardo, 1988). When a tool must go into a
full magazine, remove the one whose next use is furthest away.\\[0.6em]
Optimal, and $O(n|T|)$.}
\column{0.47\textwidth}
\textbf{Ordering is hard.}\\[0.5em]
{\small NP-hard for every fixed capacity $b \ge 2$ (Crama et al., 1994).\\[0.6em]
A good order puts jobs with similar requirements next to each other --- but ``similar''
means only \emph{overlapping sets}, with no natural distance, and the magazine limit
couples jobs that sit far apart in the order.}
\end{columns}
\vspace{1.2em}
\keyline{}
\centering
An easy inner problem inside a hard outer one --- exactly what decomposition is built for.
\end{frame}
\note{90 seconds. The whole talk hinges on this asymmetry: easy inside, hard outside.
If a juror stops you here, that is fine -- it means they are following.}

%------------------------------------------------------------------ two-phase
\begin{frame}{Nobody solves it directly}{The two-phase method, and the question nobody had answered}
\vspace{0.5em}
\begin{enumerate}\itemsep4pt
  \item \textbf{Group} the jobs into as few magazine-fitting batches as possible.
  \item \textbf{Sequence} the batches.
  \item \textbf{Re-optimise the loading} across the whole flattened order.
\end{enumerate}
\vspace{0.8em}
{\small It is fast, it is what the strongest metaheuristics start from, and it is what
industry uses.}
\vspace{1.0em}
\keyline{}
\begin{center}
\textbf{No worst-case analysis of it had been published.}\\[0.4em]
{\small How far from optimal can it be? On which instances is it exact?}
\end{center}
\end{frame}
\note{90 seconds. Emphasise step 3. Almost everyone forgets step 3, and forgetting it
is exactly the error in the literature that Result 1 corrects.}

%------------------------------------------------------------------ consequences
\begin{frame}{What follows from it}
\vspace{0.1em}\small
\begin{itemize}\itemsep3pt
  \item \textbf{Zero-gap classes for every capacity} --- families on which the two-phase
  method is provably exact.
  \item \textbf{Worst-case guarantees} better than the trivial $b(\Kst-1)$; extremal
  behaviour settled exactly at three groups.
  \item \textbf{A census.} In $420$ random instances exactly one has a positive
  heuristic gap. Every positive gap found anywhere equals \emph{one}, and every instance
  carrying one has $b\ge4$.
  \item \new{\textbf{An open question of Prof.\ Wagler's, closed.}} At $b=3$ a job is an
  \emph{edge} and a tool a \emph{vertex}, so the family is finite and can be enumerated.
  Odd rings are \emph{not} the only obstruction to integrality --- the smallest is a
  three-job \textbf{star}, which has no odd cycle at all.
\end{itemize}
\vspace{0.3em}
\keyline{}
\begin{center}
{\footnotesize The census is exhaustive on small instances. Burger et al.\ report from a
\emph{printing press} that the decomposition degrades as the magazine grows relative to
the largest job. \textbf{Two completely different methods, the same regime.}}
\end{center}
\end{frame}

\vspace{0.1em}

\vspace{0.3em}
\keyline{}
\vspace{0.1em}

a tool is a \textbf{vertex}, a job is an \textbf{edge},\\
a feasible group is a subgraph on $\le3$ vertices,\\
and the $k$-ring is the cycle $C_k$.\\[0.6em]
So the whole family can be enumerated: \textbf{151} instances up to isomorphism.}

%------------------------------------------------------------------ exact side
\begin{frame}{The exact side, and the wall it runs into}
\vspace{0.5em}
{\small Every compact formulation proposed in forty years has a weak linear relaxation.
The strongest of them relaxes to}
\[
  \new{q \;=\; |\U| - b}
\]
{\small --- every tool that does not fit in the initial magazine must be inserted at
some point. The strongest relaxation available is this counting argument.}
\vspace{0.9em}
\keyline{}
\textbf{Two methods built to get past it:}
\begin{itemize}\itemsep4pt
  \item \textbf{Branch-and-Benders-cut} --- put the ordering in a master problem and
  settle the loading exactly with the polynomial oracle, so no weak relaxation is ever
  formed.
  \item \textbf{Position-indexed branch-and-price} --- PCF$'$ prices one magazine per
  position; PTF prices \emph{pairs} of consecutive magazines, and its relaxation can
  provably exceed $q$.
\end{itemize}
\end{frame}
\note{90 seconds. The point to land: the strongest relaxation anyone has for this
problem is a one-line counting argument. Say it as a fact about the problem, not as a
criticism of the papers -- it sets up everything after it.}

%------------------------------------------------------------------ RESULT 4 diagnosis
\begin{frame}{\new{Result 4.} Why the bound will not move}
\vspace{0.1em}
Write $\beta_t$ for the number of blocks of positions in which tool $t$ sits in the magazine:
\[
  \Zst=\sum_{t\in\U}\beta_t,\qquad \beta_t\ge1 \;\Longrightarrow\; \Zst \ge q .
\]
{\small Everything above $q$ is a \textbf{re-insertion} --- a tool evicted and brought
back. That happens because of the \emph{global interleaving} of the schedule.}
\vspace{0.4em}
\keyline{}
\begin{center}
\textbf{But every inequality either method has is \new{local}.}
\end{center}
\vspace{0.1em}
\begin{itemize}\itemsep3pt\small
  \item Benders: one aggregated row over \emph{arcs}; triplet rows that are worth zero at
  a fractional point; and every optimality cut is arc-supported too.
  \item PCF$'$: one configuration per \emph{position}, coupled only per tool --- so the
  relaxation mixes fractionally and pays no transition.
  \item \textbf{Measured:} on $87.4\%$ of Benders runs the root LP equals $|\U|$ exactly.
  The pairwise and triplet rows contribute \emph{nothing}.
\end{itemize}
\end{frame}
\note{Two minutes, and this is the intellectual centre of the talk. Get the word
'local' to land. Cost is global, every inequality is local, therefore the bound cannot
move. 87.4 per cent is the evidence.}

%------------------------------------------------------------------ RESULT 5 window
\begin{frame}{\new{Result 5.} A family that is not local, and what it is worth}
\vspace{0.1em}
{\small For a stretch of positions $W=[p,r]$: every tool needed inside $W$ is either
already loaded at $p-1$, or inserted within $W$.}
\[
  \sum_{t}\sum_{k\in W} w_{t,k} \;\;\ge\;\;
  \bigl|\{t \text{ present somewhere in } W\}\bigr| \;-\; \sum_{t} a_{t,p-1}
\]
{\small Valid --- checked at $1{,}200{,}816$ integer points, no violation. And
\textbf{robust}: only master variables, so the pricing oracle is untouched.}
\vspace{0.4em}
\keyline{}
\centering
{\small Best bound each family could \emph{ever} certify, on the $62$ loose instances:}\\[0.5em]
\begin{tabular}{@{}lrrr@{}}
\toprule
family & mean & median & no gain on\\
\midrule
arc-supported --- what both solvers carry today & $8.9\%$ & $0\%$ & $46/62$\\
\new{window} & \new{$30.9\%$} & \new{$40\%$} & $18/62$\\
\bottomrule
\end{tabular}
\vspace{0.4em}

{\small The arc family is \emph{exhausted}. The window family has somewhere to go.}
\end{frame}

%------------------------------------------------------------------ negatives
\begin{frame}{Four things that did not work --- and why that is the evidence}
\vspace{0.1em}
\begin{itemize}\itemsep4pt\small
  \item \textbf{Fractional Benders cuts.} $67.5$ million generated; node count fell
  twentyfold; the bound at termination rose on \emph{no} instance; $67$ solved instances
  lost.
  \hfill{\color{cGrey}\scriptsize They are arc-supported, so they re-derive what the integer cuts already say.}
  \item \textbf{Pareto-optimal cut lifting.} $-37$ instances.
  \hfill{\color{cGrey}\scriptsize Choosing a deeper cut inside the wrong family.}
  \item \textbf{Conflict-graph row.} Neutral --- as predicted, the coverage row dominates it.
  \item \textbf{Only the primal strengthening helped} ($+31$): a hybrid genetic warm start.
  \hfill{\color{cGrey}\scriptsize A solver whose difficulty was finding solutions would not behave this way.}
  \item \textbf{Learned cut selection.} Built and trained, but on knapsack cover cuts,
  \emph{not} on the SSP. It learns --- $46$--$73\%$ over random selection --- and says
  nothing about this problem.
  \hfill{\color{cGrey}\scriptsize Selection cannot pick a strong cut that does not exist.}
\end{itemize}
\vspace{0.2em}
\keyline{}
\centering\small
These are not four separate disappointments. They are \emph{four independent
measurements of the same thing}: every one of them attacks the bound, and every one of
them fails in the way the diagnosis predicts.
\end{frame}
\note{90 seconds. Do not apologise for this slide. Four independent attacks on the
bound, four failures, all in the direction the diagnosis predicts. That is the evidence,
not a list of disappointments.}

%------------------------------------------------------------------ conclusions
\begin{frame}{Conclusions}
\vspace{0.1em}
\begin{itemize}\itemsep3pt\small
  \item The two-phase heuristic admits \textbf{two readings}, and the literature's worst
  case belongs to the wrong one. The method is better than the model of it.
  \item The gap is exactly the \textbf{price of using the fewest groups} --- with zero-gap
  classes, worst-case bounds, and a census that agrees with the one industrial study.
  \item A configuration-based exact method for the SSP is \textbf{bound-limited, not
  implementation-limited}. Established four independent ways.
  \item The bound will not move because every inequality we have is \textbf{local} and
  the cost is not. \new{A non-local family exists, is valid and robust, and has roughly
  four times the headroom.}
\end{itemize}
\vspace{0.4em}
\keyline{}
\small\textbf{Next:} measure the window family inside the solver; chain PTF per configuration
rather than per tool; disaggregate the Benders master's single $\theta$. And the open
problem --- \emph{is the excess of PTF over $q$ bounded by a constant?}
\end{frame}
\note{90 seconds, then stop talking. Leave the open problem on screen -- it invites the
first question onto ground you have prepared.}

\begin{frame}{Backup --- the two cost conventions}
\vspace{0.6em}
{\small Formulations disagree on whether the initial magazine is free. The two costs
differ by the size of the initial load, so native objectives are \textbf{not} comparable
across methods.}
\vspace{0.8em}
\begin{itemize}\itemsep4pt\small
  \item Every comparison in this work is on the \emph{empty-start} cost of the returned
  sequence, re-evaluated by the loading oracle.
  \item SSPMF reports the free-initial cost. Difference $=b$ on $1{,}029$ of its $1{,}038$
  closed instances; smaller on the other nine --- exactly those whose optimal sequence
  starts with a job needing fewer than $b$ tools.
  \item An instance must be keyed by \emph{(family, name, jobs, tools, capacity)}. Crama
  reuses file names across four capacities; keying without it manufactures disagreements
  that do not exist.
  \item Ten instances declare a tool that no job requires, so the bound is $|\U|-b$ and
  not $|T|-b$.
\end{itemize}
\end{frame}

\begin{frame}{Backup --- the window inequality}
\vspace{0.5em}
{\small For a stretch of positions $W=[p,r]$, every tool needed inside $W$ is either
already loaded at $p-1$ --- where the magazine holds exactly $b$ --- or inserted within
$W$:}
\[
  \sum_{t}\sum_{k\in W} w_{t,k} \;\;\ge\;\;
  \bigl|\{t \text{ present somewhere in } W\}\bigr| \;-\; \sum_{t} a_{t,p-1}
\]
\vspace{-0.4em}
\keyline{}
\begin{itemize}\itemsep3pt\small
  \item \textbf{Proved}, not merely tested. One auxiliary variable per tool and window
  linearises it.
  \item \textbf{Robust}: only master variables appear, so its dual never reaches the
  pricing problem. The set-union knapsack oracle is untouched.
  \item The per-tool counting rows already in the master are the case $W=[2,n]$, so this
  \emph{generalises} what is there.
  \item $84\%$ of the windows that carry the bound have length $\le 4$, so a static pool
  suffices --- no separation routine needed.
\end{itemize}
\end{frame}
