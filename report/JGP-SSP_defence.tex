\begin{frame}{Drop the fewest-groups rule: grouping is exact}

\begin{frame}{The loading LP is integral on a fixed path}
\begin{columns}[T,onlytextwidth]
\column{.61\textwidth}
\small
\[
\begin{aligned}
\min\quad &\sum_{j,t}z_{jt}\\
\text{s.t.}\quad
&y_{jt}\ge1 &&(t\in T_j),\\
&\sum_ty_{jt}\le b &&(j\in J),\\
&z_{jt}\ge y_{jt}-y_{it}-(1-x_{ij}) &&(i\ne j),\\
&z_{jt}\ge y_{jt}-(1-x_{dj}) &&(j\in J),\\
&y,z\ge0.
\end{aligned}
\]

\column{.35\textwidth}
\vspace{.18cm}
$y_{jt}$ records whether tool $t$ is loaded at job $j$.

\vspace{.28cm}
$z_{jt}$ charges an insertion when the selected path moves into $j$.

fixed path}\par
{\large $\Downarrow$}\par
{\fontsize{18}{22}\selectfont exact KTNS cost}
\end{columns}
