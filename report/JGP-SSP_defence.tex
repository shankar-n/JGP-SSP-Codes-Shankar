\begin{frame}{Two questions, and how the talk is organised}
\vspace{.15cm}
\begin{columns}[T,onlytextwidth]
\column{.47\textwidth}
\theory{Question 1 \; -- \; structural}\par\vspace{.16cm}
Industry does not solve the SSP. It groups jobs into as few magazine-fitting batches

Every compact model surveyed relaxes to the counting bound $q$, so it is fast where

\textbf{Can a method that never forms that relaxation} -- an exact loading oracle, or a
reformulation over configurations -- get past it?
\end{columns}

\begin{block}{Where the loss comes from}
Allowing \emph{all} feasible groupings makes the construction \textbf{exact}: the optimum
equals the cheapest walk over every grouping. So the loss is not a defect of the method
--- it is the price of insisting on the \textbf{fewest} groups.
\end{block}

\vspace{.35cm}
\centering\small
Two readings must be separated: the walk through one fixed magazine per group, and the
value after KTNS re-optimises. \textbf{They differ.}
\cit{Crama \& van de Klundert, 1999}
\end{frame}

For every $g\ge1$ there is a \textbf{connected} instance with $b=6$,

\begin{frame}{Branch-and-Benders-cut: the idea}
\vspace{.35cm}
\centering
\begin{tikzpicture}[x=1cm,y=1cm,>=Stealth,
 blk/.style={draw=navy!40,rounded corners=2pt,minimum width=3.6cm,minimum height=1.35cm,
             align=center,font=\small}]
  \node[blk,fill=warm] (m) at (0,0) {\textbf{Master} \; (hard)\\choose the job order\\minimise the stand-in $\theta$};
  \node[blk,fill=paleorange,draw=orange!70] (s) at (6.4,0) {\textbf{Subproblem} \; (easy)\\load the tools for that\\order, \emph{exactly}};
  \draw[->,navy,line width=1.1pt] (m.north east) to[bend left=18] node[above,font=\scriptsize] {an order $\bar x$} (s.north west);
  \draw[->,orange,line width=1.1pt] (s.south west) to[bend left=18] node[below,font=\scriptsize,align=center] {a cut: $\theta\ge$ true cost} (m.south east);
\end{tikzpicture}

\vspace{.75cm}
\begin{block}{Why this problem invites Benders}
For a \textbf{fixed} order the loading is solvable in polynomial time. So the subproblem
is not relaxed --- it is solved exactly, and the master's bound is built from true
subproblem values rather than from a weak relaxation of the loading.
\end{block}
\cit{Hooker \& Ottosson, 2003; Codato \& Fischetti, 2006; Rahmaniani et al., 2017}
\end{frame}


\begin{frame}{The master problem}
\vspace{.4cm}
Arc variables $x_{ij}\in\{0,1\}$ on jobs plus a depot $d$ marking start and end, so the
$x$ describe a Hamiltonian path.
\vspace{.35cm}
\[
\min\;\theta
\qquad\text{s.t.}\qquad
\sum_{i\ne j}x_{ij}=1\;\;(\forall j),\qquad
\sum_{j\ne i}x_{ij}=1\;\;(\forall i),
\]
\[
\sum_{i,j\in S}x_{ij}\le\lvert S\rvert-1 \quad (S\subseteq J),
\qquad
\theta\;\ge\;\lvert\U\rvert .
\]

\vspace{.45cm}
\begin{block}{Two bounds the master starts from}
The \textbf{coverage row} $\theta\ge\lvert\U\rvert$, and a \textbf{pairwise row}
$\theta\ \ge\ \sum_{i,j}w_{ij}x_{ij}+\sum_j\lvert T_j\rvert x_{dj}$ with
$w_{ij}=\max\bigl(0,\lvert T_i\cup T_j\rvert-b\bigr)$.
\end{block}
\end{frame}


\begin{frame}{The subproblem: loading a fixed order}
\vspace{.3cm}
Fix the order $\bar x$. Let $y_{j,t}=1$ if tool $t$ is in the magazine while job $j$
runs, and $z_{j,t}$ charge an insertion of $t$ at $j$.
\vspace{.2cm}
{\small
\[
\begin{aligned}
\min\;\; & \textstyle\sum_{j,t} z_{j,t} &&\\
 & y_{j,t}\ \ge\ 1 && t\in T_j \quad (\nu_{jt})\\
 & \textstyle\sum_t y_{j,t}\ \le\ b && \forall j \quad (\mu_j)\\
 & z_{j,t}\ \ge\ y_{j,t}-y_{i,t}-(1-\bar x_{ij}) && \forall i\ne j,\ t \quad (\lambda_{ijt})\\
 & z_{j,t}\ \ge\ y_{j,t}-(1-\bar x_{dj}) && \forall j,\ t \quad (\lambda^{d}_{jt})
\end{aligned}
\]}
\vspace{-.15cm}
\begin{block}{Why the oracle is exact, not a price}
For fixed $\bar x$ the master variables sit only on the right-hand side. The remaining
matrix has the consecutive-ones property along the selected path and is totally
unimodular, so the relaxation is \textbf{integral} and its optimum is exactly the KTNS
cost of that order.
\end{block}
\cit{Tang \& Denardo, 1988}
\end{frame}


\begin{frame}{The Benders cut, in full}
\vspace{.4cm}
Only $\bar x$ appears on the right-hand side, so in the dual it appears only in the
objective. Taking an optimal dual $(\bar\lambda,\bar\lambda^{d},\bar\mu,\bar\nu)$:
\vspace{.3cm}
\[
\boxed{\;\;
\theta\;\ge\;
\sum_{i\ne j}\sum_{t}\bar\lambda_{ijt}\,(x_{ij}-1)
\;+\;\sum_{j}\sum_{t}\bar\lambda^{d}_{jt}\,(x_{dj}-1)
\;-\;b\sum_{j}\bar\mu_j
\;+\;\sum_{j}\sum_{t\in T_j}\bar\nu_{jt}\;\;}
\]

\vspace{.5cm}
\begin{columns}[T,onlytextwidth]
\column{.47\textwidth}
\textbf{Valid globally.} The dual feasible region does not depend on $\bar x$, so the
cut holds for every order.
\column{.47\textwidth}
\textbf{Tight locally.} At $\bar x$ it reproduces the exact loading cost. It is affine
in $x$, so fractional separation needs no separate argument.
\end{columns}
\end{frame}


\begin{frame}{One term, and 193 lost optima}
\vspace{.5cm}
\begin{block}{A defect found during the campaign}
The depot sum $\sum_{j,t}\bar\lambda^{d}_{jt}(x_{dj}-1)$ was initially omitted.
The cut then became \textbf{too tight}: it cut off genuine optima.
\end{block}

\vspace{.6cm}
\begin{columns}[T,onlytextwidth]
\column{.47\textwidth}
\textbf{How it surfaced}\\[.25cm]
{\small Randomly generated instances were solved by brute force and by the solver.
\textbf{193} violations of the true optimum were recorded.}
\column{.47\textwidth}
\textbf{What was done}\\[.25cm]
{\small The term was restored, the derivation was rewritten from the implementation,
and the entire campaign was re-run.}
\end{columns}

\vspace{.6cm}
\centering\small
The first magazine is charged in full because the machine starts empty --- dropping it
silently assumes the first job's tools are free.
\end{frame}


\begin{frame}{Two cut families, and what the ablation tests}
\vspace{.4cm}
\begin{columns}[T,onlytextwidth]
\column{.47\textwidth}
\textbf{\color{navy}LP dual cuts} \; (\texttt{BBC-LP})\\[.2cm]
{\small The cut above, from an optimal dual of the loading LP at an integer master
solution.}
\column{.47\textwidth}
\textbf{\color{navy}Combinatorial cuts} \; (\texttt{BBC-K})\\[.2cm]
{\small A no-good style cut built directly from the KTNS value of the proposed order.}
\end{columns}

\vspace{.7cm}
\begin{block}{Four strengthenings, each a standard remedy}
\begin{itemize}\itemsep2pt
  \item[\texttt{+F}] \textbf{fractional separation} --- cut at fractional master points, not only integer ones
  \item[\texttt{+T}] \textbf{triplet bounds} --- a three-job strengthening of the pairwise row
  \item[\texttt{+C}] \textbf{conflict-graph row} --- jobs that cannot share a magazine
  \item[\texttt{+P}] \textbf{Pareto-optimal lifting} --- choose the deepest among alternative optimal duals
  \item[\texttt{+H}] \textbf{hybrid incumbent} --- a genetic heuristic run at the root
\end{itemize}
\end{block}
\end{frame}


\begin{frame}{Fractional separation: the remedy that should have worked}
\vspace{.35cm}
\begin{columns}[T,onlytextwidth]
\column{.46\textwidth}
\textbf{The reasoning}\\[.25cm]
{\small When a Benders scheme tails off, the standard fix is to separate at
\emph{fractional} points rather than waiting for integer ones. The cut is affine in
$x$, so this is legitimate.}
\vspace{.45cm}

\textbf{It did what it should, locally}\\[.25cm]
{\small $67.5$ million cuts generated. Median node count fell from
$\mathbf{77{,}439}$ to $\mathbf{3{,}701}$ --- a factor of twenty.}

\column{.48\textwidth}
\begin{block}{And the bound never moved}
Across $1{,}404$ paired runs, the dual bound at termination was
\vspace{.15cm}
\begin{center}
\begin{tabular}{@{}lr@{}}
higher & \textbf{0}\\
equal  & $1{,}226$\\
lower  & $178$\\
\end{tabular}
\end{center}
\vspace{.1cm}
and $\mathbf{69}$ instances were \emph{lost}.
\end{block}
\vspace{.3cm}
{\small The remedy is applied correctly, to the wrong obstacle.}
\end{columns}
\end{frame}

\begin{frame}{Backup --- Branch-and-price: why reformulate at all}
\vspace{.45cm}
\begin{block}{The obstacle}
Every compact model surveyed relaxes to the counting bound $q=\lvert\U\rvert-b$ or
below. A branch-and-cut built on any of them is fast where $q$ is exact and adrift
where it is not.
\end{block}

\vspace{.5cm}
Reformulating over \textbf{magazine configurations} attacks this from the other side.
A configuration is a set $C\subseteq\U$ with $\lvert C\rvert=b$; these become the
columns, generated on demand. Three formulations were built.

\vspace{.5cm}
\centering
\begin{tabular}{@{}lll@{}}
\toprule
& \textbf{Column} & \textbf{Linear relaxation}\\
\midrule
PCF        & configuration at a position & $=q$ with the counting rows, $0$ without\\
PCF$'$     & prefix form of PCF          & $=q$, symmetry removed\\
PTF        & \emph{transition} between two & $\ \ge q$, and can exceed it\\
\bottomrule
\end{tabular}
\end{frame}


\begin{frame}{Backup --- PCF: configurations at positions}
\vspace{.3cm}
$y_C^{k}=1$ if configuration $C$ occupies position $k$; $w_t^{k}\ge0$ counts an
insertion of $t$ at position $k$.
\vspace{.25cm}
{\small
\[
\begin{aligned}
\min\;\; & \textstyle\sum_{t,k}w_t^{k} &&\\
\text{(P)}\;\; & \textstyle\sum_C y_C^{k}=1, & k&=1,\dots,n\\
\text{(Cov)}\; & \textstyle\sum_k\sum_{C\supseteq T_j}y_C^{k}\ \ge\ 1, & j&\in J\\
\text{(W)}\;\; & w_t^{k}\ \ge\ \textstyle\sum_{C\ni t}\bigl(y_C^{k}-y_C^{k-1}\bigr), & t&\in T,\ k\ge2\\
\text{(T)}\;\; & \textstyle\sum_{k\ge2}w_t^{k}+\sum_{C\ni t}y_C^{1}\ \ge\ 1, & t&\in\U
\end{aligned}
\]}
\vspace{-.1cm}
\begin{block}{The counting rows are load-bearing}
Without (T) the relaxation is \textbf{zero} --- one convex blend of configurations,
repeated at every position, satisfies (P), (Cov) and (W) at no cost. With (T) it is
exactly $q$. \emph{Proved, and confirmed on all $2{,}108$ recorded roots.}
\end{block}
\end{frame}


\begin{frame}{Backup --- PTF: transitions as columns}
\vspace{.3cm}
$z_{C,C'}^{k}\ge0$ is a column for the step from position $k$ to $k+1$ leaving $C$ and
entering $C'$, at cost $d(C,C')=\lvert C'\setminus C\rvert$.
\vspace{.2cm}
{\small
\[
\begin{aligned}
\min\;\; & \textstyle\sum_{k}\sum_{(C,C')}d(C,C')\,z_{C,C'}^{k} &&\\
\text{(S)}\;\; & \textstyle\sum_{(C,C')}z_{C,C'}^{k}=1, & k&=1,\dots,n-1\\
\text{(F)}\;\; & \textstyle\sum_{(C,C'):t\in C'}z^{k}_{C,C'}=\sum_{(D,D'):t\in D}z^{k+1}_{D,D'}, & t&\in T\\
\text{(Cov)}\; & \text{every job served at the head or tail of some step} &&\\
\text{(T)}\;\; & \textstyle\sum_{k}\sum_{(C,C'):t\notin C,\,t\in C'}z^{k}_{C,C'}+\sum_{(C,C'):t\in C}z^{1}_{C,C'}\ \ge\ 1, & t&\in\U
\end{aligned}
\]}
\vspace{-.1cm}
\begin{block}{Why this one can beat the counting bound}
A column \emph{carries a step}, so the relaxation cannot mix configurations without
paying the transition. Hence $\mathrm{LP}(\mathrm{PTF})\ge q$, and it can be strictly
greater --- the only construction in this work that provably exceeds $q$.
\end{block}
\end{frame}


\begin{frame}{Backup --- pricing, and what the prototypes measured}
\vspace{.4cm}
\begin{columns}[T,onlytextwidth]
\column{.46\textwidth}
\textbf{The pricing problem}\\[.25cm]
{\small Choosing the entering configuration is \textbf{NP-hard}: pick $b$ tools to
collect per-tool duals plus a bonus for each job fully covered. PTF prices
\emph{coupled pairs}, so a node costs far more again.}
\vspace{.4cm}

\textbf{Rows stay small}\\[.25cm]
{\small $O(n\lvert T\rvert)$ and fixed; only the columns are generated.}

\column{.48\textwidth}
\begin{block}{What was measured}
\begin{itemize}\itemsep2pt\small
  \item PCF$'$ root $=q$ on \textbf{all $2{,}108$} recorded runs --- the proposition
  confirmed on the benchmark.
  \item PTF exceeds $q$ on \textbf{2 of 232} roots, rising from $5$ to $6$ where the
  optima are $12$ and $13$.
  \item Nothing above nine jobs was closed: a Python pricer against compiled solvers.
\end{itemize}
\end{block}
\vspace{.25cm}
{\small Sound, and as it stands too rare and too small to close the certificate gap.}
\end{columns}
\end{frame}
