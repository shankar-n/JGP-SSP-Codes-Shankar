\small\textbf{Scope and Status.}
This document analyses the \emph{JGP~+~GSP decomposition heuristic}:
solve the Job Grouping Problem (JGP) to get groups, then for each group select
a configuration and solve a Group Sequencing Problem (GSP/GTSP) to find the
best ordering and configurations jointly.
We establish:
(i)~the JGP formulation and its relationship to the SSP objective;
(ii)~a verified counterexample showing the gap is non-zero;
(iii)~classical bounds ($\ZJGP \le \ZSSP + 1$ and $\ZSSP \le b \cdot \ZJGP$);
(iv)~a construction showing the gap grows without bound in $K^*$.
\textbf{Open problem:} a tight bound for the gap when both $K^*$ and $b$ are
fixed. This is marked explicitly as \textbf{TODO}.
\end{mdframed}

\section{The JGP + GSP Heuristic}
\label{sec:heuristic}
% ─────────────────────────────────────────────────────────────────────────────

\subsection{The Job Grouping Problem (JGP)}

\begin{definition}[Feasible group]
A subset of jobs $G \subseteq J$ is \emph{feasible} if all their tools fit
simultaneously into the magazine:
$\bigl|\bigcup_{j \in G} T_j\bigr| \le b$.
\end{definition}

The JGP asks for the minimum number of feasible groups needed to partition $J$:

\begin{definition}[JGP as a set covering problem \cite{Crama1994}]
Let $\mathcal{S}$ be the family of all feasible groups.
Introduce binary variable $y_G \in \{0,1\}$ (1 if group $G$ is used).
\begin{align}
  \min \quad & \sum_{G \in \mathcal{S}} y_G \label{eq:jgp_obj}\\
  \text{s.t.}\quad
  & \sum_{G \in \mathcal{S}:\, j \in G} y_G \ge 1 \quad \forall j \in J
    \label{eq:jgp_cover}\\
  & y_G \in \{0,1\} \quad \forall G \in \mathcal{S} \notag
\end{align}
The optimal value $\ZJGP$ is the minimum number of groups (``batches'') needed.
\end{definition}

\begin{remark}[JGP hardness]
The JGP is NP-hard \cite{Crama1994} (it generalises graph colouring).
Practical solution uses column generation: the pricing subproblem generates
maximal feasible groups, equivalent to a bin-packing variant. Felipe Otiai's
thesis \cite{Otiai2024} develops an exact Branch-and-Price solver for the JGP.
\end{remark}

\subsection{From Groups to Configurations}

Once the JGP yields groups $G_1, \ldots, G_K$ (where $K \ge \ZJGP$), each group
must be equipped with a \emph{configuration} $C_k$: a maximal magazine state that
covers all the group's requirements.

\begin{definition}[Configuration cluster for a group]
For group $G$, the set of all valid configurations is:
\[
  \mathcal{C}(G) = \Bigl\{ C \subseteq T \;\Big|\; |C|=b,\;
  \textstyle\bigcup_{j \in G} T_j \subseteq C \Bigr\}.
\]
The slack of group $G$ is $s(G) = b - |\bigcup_{j\in G}T_j| \ge 0$;
this is exactly the number of ``free'' padding slots.
\end{definition}

The padding slots in $C_k$ are the key degree of freedom: they can be filled
with tools from \emph{adjacent groups} to reduce the inter-group switching cost.
This padding choice is central to the gap analysis.

\subsection{The Group Sequencing Problem (GSP)}

Given $K$ groups and their configuration clusters, the \emph{Group Sequencing
Problem (GSP)} jointly selects one configuration per group and finds the optimal
ordering of the groups to minimise total switching cost.

Formally, given groups $G_1,\ldots,G_K$, choose $C_k \in \mathcal{C}(G_k)$ for
each $k$ and a permutation $\pi$ of $\{1,\ldots,K\}$ to minimise:
\[
  H(\pi, \{C_k\}) = \sum_{i=1}^{K-1}
  d\!\left(C_{\pi(i)},\, C_{\pi(i+1)}\right)
  = \sum_{i=1}^{K-1} \bigl|C_{\pi(i+1)} \setminus C_{\pi(i)}\bigr|.
\]

This is a \textbf{Generalised TSP (GTSP)} on clusters $\{\mathcal{C}(G_k)\}$
with arc costs inherited from the SSP metric $d$. For a fixed ordering $\pi$,
the optimal configuration selection can be found greedily (lookahead padding,
\cref{sec:lookahead}). Over all orderings, the problem is a GTSP and is solved
by standard GTSP/TSP solvers (Concorde, LKH-3, Noon-Bean transformation).

\begin{definition}[JGP + GSP heuristic]
\label{def:heuristic}
Given a U-SSP instance:
\begin{enumerate}
\item Solve the JGP (exactly or approximately) to obtain groups $G_1,\ldots,G_K$.
\item Solve the GSP (GTSP) over clusters $\{\mathcal{C}(G_k)\}$ to find the
  optimal permutation $\pi^*$ and configurations $\{C_k^*\}$.
\item The resulting switching cost $H = H(\pi^*, \{C_k^*\})$ is a valid upper
  bound for the SSP: $H \ge \ZSSP$.
\end{enumerate}
\end{definition}

\begin{warnbox}
\textbf{Important distinction.}
The GSP optimises over \emph{all valid configurations} for each group,
not just one fixed ``canonical'' choice. Any analysis that fixes the padding
to a single strategy (e.g., all padding tools disjoint from adjacent groups)
is studying a specific sub-optimal variant of the heuristic, not the JGP+GSP
heuristic as defined above.
\end{warnbox}

\subsection{Lookahead Configurations Achieve the GSP Optimum}
\label{sec:lookahead}

For a \emph{fixed} ordering $\pi$ of the groups, the optimal configuration for
group $G_{\pi(k)}$ is obtained by \emph{lookahead}: fill the $s(G_{\pi(k)})$
padding slots with as many tools from the next group $G_{\pi(k+1)}$ as possible.

\begin{thmbox}
\begin{lemma}[Achievable per-transition cost]
\label{lem:transition}
For fixed adjacent groups $G_k$, $G_{k+1}$ (in that order), the minimum
switching cost achievable by any choice of configurations is:
\[
  w^*(G_k, G_{k+1})
  = \max\!\Bigl(0,\;
  \bigl|\textstyle\bigcup_{j \in G_k} T_j \setminus \bigcup_{j \in G_{k+1}} T_j\bigr|
  - s(G_{k+1})\Bigr)
  = \max\!\Bigl(0,\;
  \bigl|\textstyle\bigcup_{j \in G_k} T_j \cup \bigcup_{j \in G_{k+1}} T_j\bigr| - b
  \Bigr).
\]
This lower bound is achieved by the \emph{lookahead configuration}:
$C_k^* = \bigl(\bigcup_{j\in G_k}T_j\bigr) \cup X_k$ where
$X_k \subseteq \bigl(\bigcup_{j\in G_{k+1}}T_j\bigr) \setminus
\bigl(\bigcup_{j\in G_k}T_j\bigr)$ and $|X_k| = \min\bigl(s(G_k),\,
|\bigl(\bigcup_{j\in G_{k+1}}T_j\bigr)\setminus\bigl(\bigcup_{j\in G_k}T_j\bigr)|\bigr)$.
\end{lemma}
\end{thmbox}

\begin{proofbox}
\begin{proof}
Denote $T^{(k)} = \bigcup_{j \in G_k}T_j$. Configuration $C_{k+1}$ must contain
$T^{(k+1)}$ (mandatory, $|T^{(k+1)}|$ slots) and has $s_{k+1} = b - |T^{(k+1)}|$
free slots. Of the $|T^{(k)} \setminus T^{(k+1)}|$ tools in $C_k$ not needed by
$G_{k+1}$, at most $s_{k+1}$ can be retained in $C_{k+1}$ as padding. The remaining
$\max(0, |T^{(k)} \setminus T^{(k+1)}| - s_{k+1})$ tools must leave, each costing
one switch. The second equality follows from $|T^{(k)} \setminus T^{(k+1)}| - s_{k+1}
= |T^{(k)} \cup T^{(k+1)}| - b$. The lookahead configuration pre-loads as many
next-group tools as possible in the padding slots of $C_k$, maximising the tools
that survive into $C_{k+1}$. \qed
\end{proof}
\end{proofbox}

\section{JGP vs.\ SSP: Fundamental Relationship}
\label{sec:relationship}
% ─────────────────────────────────────────────────────────────────────────────

The JGP optimal value $\ZJGP$ and the SSP optimal value $\ZSSP$ are related
by two classical bounding theorems.

\begin{thmbox}
\begin{theorem}[Lower bound: $\ZJGP - 1 \le \ZSSP$ \cite{Crama1994}]
\label{thm:lb}
The optimal SSP cost is at least $\ZJGP - 1$.
\end{theorem}
\end{thmbox}

\begin{proofbox}
\begin{proof}
Let $\sigma^*$ be the optimal SSP sequence, traversing $m$ distinct maximal
configurations $W_1, W_2, \ldots, W_m$ (in order of first use). Since this
sequence processes all jobs and each $W_r$ is a valid configuration,
$\{W_1, \ldots, W_m\}$ is a valid cover of $J$, so $m \ge \ZJGP$.
Since consecutive distinct configurations differ in at least one tool,
each of the $m-1$ transitions costs $\ge 1$ switch:
$\ZSSP \ge m-1 \ge \ZJGP - 1$. \qed
\end{proof}
\end{proofbox}

\begin{thmbox}
\begin{theorem}[Upper bound: $\ZSSP \le b \cdot \ZJGP$ \cite{Crama1994}]
\label{thm:ub}
The optimal SSP cost is at most $b$ times the optimal JGP value.
\end{theorem}
\end{thmbox}

\begin{proofbox}
\begin{proof}
Take any JGP-optimal partition into $K^* = \ZJGP$ groups $G_1,\ldots,G_{K^*}$.
Process jobs batch by batch. The worst-case transition between any two consecutive
groups occurs when their configurations are completely disjoint, requiring $b$
tool insertions. With $K^* - 1$ transitions and an initial load of $\le b$ tools,
total switches $\le b + b(K^*-1) = b \cdot K^* = b \cdot \ZJGP$.
Since $\ZSSP$ is the minimum, $\ZSSP \le b \cdot \ZJGP$. \qed
\end{proof}
\end{proofbox}

\begin{remark}[Are the bounds tight?]
The lower bound $\ZJGP - 1 \le \ZSSP$ is tight (e.g., any instance where the
SSP processes each group as a single configuration with zero intra-group switches
achieves $\ZSSP = \ZJGP - 1$). The upper bound $\ZSSP \le b \cdot \ZJGP$ is
extremely loose for structured instances; improving it is a primary research goal.
\end{remark}

\section{A Verified Counterexample: SSP Optimal $\ne$ JGP Optimal Configurations}
\label{sec:counterexample}
% ─────────────────────────────────────────────────────────────────────────────

The following example demonstrates that (a)~the configurations used in an optimal
SSP solution need not come from a JGP-optimal grouping, and (b)~even with the best
JGP-optimal grouping, the JGP+GSP heuristic may not achieve $\ZSSP$.

\begin{example}[6-job ring instance]
\label{ex:ring}
Let $T = \{t_1,\ldots,t_6\}$, $b = 3$, and six jobs with requirements:
\[
  T_{j_1}=\{t_1,t_2\}, \quad T_{j_2}=\{t_2,t_3\}, \quad T_{j_3}=\{t_3,t_4\},
  \quad T_{j_4}=\{t_4,t_5\}, \quad T_{j_5}=\{t_5,t_6\}, \quad T_{j_6}=\{t_6,t_1\}.
\]
This is a ``ring'' topology: consecutive jobs share exactly one tool.
The conflict graph has $|T_{j_i} \cup T_{j_k}| = 3 = b$ for adjacent pairs and
$|T_{j_i} \cup T_{j_k}| \ge 3$ for non-adjacent pairs, forcing every pair of
jobs needing disjoint tools into different groups.

\medskip
\noindent\textbf{JGP optimal solution ($\ZJGP = 3$).}
The minimum number of feasible groups is 3 (verified by PORTA enumeration).
Two optimal groupings exist; one example:
\[
  G_1 = \{j_1, j_2\},\quad G_2 = \{j_3, j_4\},\quad G_3 = \{j_5, j_6\}.
\]
Mandatory tool sets: $T^{(1)}=\{t_1,t_2,t_3\}$, $T^{(2)}=\{t_3,t_4,t_5\}$,
$T^{(3)}=\{t_1,t_5,t_6\}$. Each has slack $s=0$, so configurations are unique:
$C_1=\{t_1,t_2,t_3\}$, $C_2=\{t_3,t_4,t_5\}$, $C_3=\{t_1,t_5,t_6\}$.

\medskip
\noindent\textbf{JGP+GSP result.}
With zero slack, there is only one configuration per group; the GSP just finds
the best ordering. In any ordering, each pair of adjacent configurations
(under the cyclic ring structure) shares exactly one tool, giving $d=b-1=2$ per
transition. The best ordering has 2 transitions: total cost $= 2 \times 2 = 4$.

\medskip
\noindent\textbf{SSP optimal solution ($\ZSSP = 3$).}
Sequence $j_1 \to j_2 \to j_3 \to j_4 \to j_5 \to j_6$, processed continuously
via KTNS with $b=3$:
\begin{center}
\begin{tabular}{lccc}
\toprule
Step & Job & Magazine & Switches \\
\midrule
1 & $j_1$ & $\{t_1, t_2, t_3\}$ & 3 (initial load) \\
2 & $j_2$ & $\{t_1, t_2, t_3\}$ & 0 (no change) \\
3 & $j_3$ & $\{t_1, t_3, t_4\}$ & 1 ($-t_2, +t_4$; $t_1$ retained) \\
4 & $j_4$ & $\{t_1, t_4, t_5\}$ & 1 ($-t_3, +t_5$; $t_1$ retained) \\
5 & $j_5$ & $\{t_1, t_5, t_6\}$ & 1 ($-t_4, +t_6$; $t_1$ retained) \\
6 & $j_6$ & $\{t_1, t_5, t_6\}$ & 0 (no change) \\
\bottomrule
\end{tabular}
\end{center}
Total (excluding initial load): $\ZSSP = 3$.
KTNS retains $t_1$ as a ``bridge'' tool across jobs $j_2$--$j_5$, smoothly rolling
the ring. The JGP with 3 groups severs this bridge at both group boundaries.

\medskip
\noindent\textbf{Gap.} JGP+GSP gives $H = 4$, SSP optimum is $\ZSSP = 3$.
\textbf{Gap $= 1$}.
\end{example}

\section{Gap Growth: Constructing Unboundedly Large Differences}
\label{sec:gap_growth}
% ─────────────────────────────────────────────────────────────────────────────

The 6-job ring example achieves a gap of 1 with $K^* = 3$ groups. We now
show the gap can be made as large as desired.

\subsection{Copy-Paste Construction}

\begin{thmbox}
\begin{theorem}[Gap grows with $K^*$]
\label{thm:unbounded}
For any integer $g \ge 1$, there exists a U-SSP instance with $\ZJGP = 3g$ groups
where the JGP+GSP heuristic costs at least $g$ more switches than $\ZSSP$.
\end{theorem}
\end{thmbox}

\begin{proofbox}
\begin{proof}
Take $g$ independent copies of the 6-job ring instance from Example~\ref{ex:ring},
using completely disjoint tool sets (e.g., copy $\ell$ uses tools
$\{t_{6\ell-5}, \ldots, t_{6\ell}\}$). Process all copies in sequence.

Each copy independently has $Z_{\mathrm{JGP}}^{*(\ell)}=3$ and $Z_{\mathrm{SSP}}^{*(\ell)}=3$, with gap 1.
Since the copies share no tools, configurations from different copies are fully
disjoint and the inter-copy transitions cost exactly $b$ switches regardless.

\textbf{JGP+GSP:} The $g$ copies produce $3g$ groups total. Within each copy,
the GSP pays 4 switches (the per-copy gap is 1). Across copies, transitions cost
$b=3$ each (fully disjoint configurations), giving $g-1$ inter-copy transitions
of cost 3. Total: $H = 4g + 3(g-1)$.

\textbf{SSP optimum:} $\ZSSP = 3g + 3(g-1)$ (3 switches per copy, $b=3$ between copies).

\textbf{Gap:} $H - \ZSSP = (4g + 3(g-1)) - (3g + 3(g-1)) = g$.

Since $g$ is arbitrary, the gap is unbounded. \qed
\end{proof}
\end{proofbox}

\begin{remark}
In this construction, $K^* = 3g$ grows with $g$. The gap is $g = K^*/3$.
The gap grows \emph{proportionally to $K^*$}, suggesting the relevant quantity
for bounding the gap is $K^*$ itself, not just $b$.
This aligns with the loose upper bound: $\ZSSP \le b \cdot K^*$, so the
gap $H - \ZSSP \le H \le b \cdot K^*$.
\end{remark}

\subsection{Structure of the Gap: Two Sources}

The gap between $H$ (JGP+GSP cost) and $\ZSSP$ has two independent sources:

\begin{enumerate}
\item \textbf{Padding-choice gap:} Even with the JGP-optimal partition into
  $K^*$ groups, the GSP may not achieve $\ZSSP$ because the best configurations
  for those groups (with lookahead padding) may still cost more than the
  SSP optimal (which can choose any sequence, including ones that dissolve
  group boundaries).
\item \textbf{Grouping-choice gap:} The SSP optimal may use more than $K^*$ groups
  (e.g., $K^*+1$ groups with smaller mandatory sets), allowing configurations with
  more slack and hence cheaper transitions.
\end{enumerate}

In the 6-job ring, source (2) is zero (using 4 groups does not help), and source (1)
accounts for the full gap of 1. In the copy-paste construction, both sources contribute.

\subsection{Mandatory Exit Lower Bound (Isolating Source 1)}

For instances where the grouping-choice gap (source 2) is zero, the following
lower bound on $\ZSSP$ is tight for the specific instance family.

\begin{definition}[Mandatory exit tools]
For a JGP-optimal grouping $\{G_1,\ldots,G_{K^*}\}$ ordered as
$G_1 \to \cdots \to G_{K^*}$, define the \emph{mandatory exit set} at
boundary $k$ as:
\[
  D_k = \Bigl(\bigcup_{j \in G_k}T_j\Bigr) \setminus
        \Bigl(\bigcup_{i > k}\bigcup_{j \in G_i}T_j\Bigr),
  \quad k = 1, \ldots, K^*-1.
\]
These are tools required by $G_k$ that are never needed again in any later group.
\end{definition}

\begin{thmbox}
\begin{lemma}[Mandatory exit lower bound]
\label{lem:mandatory_exit}
For any feasible SSP solution (any grouping, any ordering), the total number of
tool switches is at least $\sum_{k=1}^{K^*-1}|D_k|$.
\end{lemma}
\end{thmbox}

\begin{proofbox}
\begin{proof}
Fix any tool $t \in D_k$. By definition, $t$ is required by some job $j \in G_k$
and is never required by any job in $G_{k+1},\ldots,G_{K^*}$.
\begin{itemize}
\item $t$ must be in the magazine when $j$ is processed (either from the
  initial load or via an insertion before $j$).
\item After the last job requiring $t$ is processed, $t$ will eventually be
  evicted (since subsequent jobs require tools not currently loaded, and the
  magazine has finite capacity $b$).
\item This eviction corresponds to at least one \emph{removal event}. In the
  uniform SSP model, $|C_u|=|C_v|=b$ for all transitions, so removals = insertions
  per transition, and each removal event corresponds to exactly one switch.
\end{itemize}
Since the sets $D_1,\ldots,D_{K^*-1}$ are pairwise disjoint (by the
``never needed again'' definition), the removal events are distinct, giving
at least $\sum_k|D_k|$ total switches. \qed
\end{proof}
\end{proofbox}

\begin{remark}[Application]
For the sliding-window construction of \texttt{06\_grouping\_selection.tex},
$|D_k| = r - \mathrm{ov}$ for all $k$ (where $r$ is the group tool set size and
$\mathrm{ov}$ is the overlap between adjacent groups), giving a lower bound of
$(K^*-1)(r-\mathrm{ov})$ on $\ZSSP$. Paired with the lookahead upper bound
of the same value, this shows $\ZSSP = (K^*-1)(r-\mathrm{ov})$ for that family.
\end{remark}

\section{Open Problems and TODO}
\label{sec:open}
% ─────────────────────────────────────────────────────────────────────────────

\begin{openproblem}
\textbf{TODO 1: Tight bound for fixed $K^*$ and $b$.}

\medskip
The copy-paste construction shows the gap $H - \ZSSP$ can be made proportional
to $K^*$. The copy-paste construction uses $g$ independent sub-instances, each
with its own 3 optimal groups, so $K^* = 3g$. But it is an open question whether,
for a \emph{fixed} $K^*$ (e.g., $K^* = 3$) and a \emph{fixed} $b$, the gap can
exceed 1.

\medskip
\textbf{Progress:} Computational experiments via PORTA on small instances suggest
the gap is at most 1 when $K^* = 3$, but no proof exists.
The goal is to answer:
\begin{quote}
\emph{For fixed $K^*$ and $b$, what is the maximum possible gap between the
JGP+GSP heuristic (with optimal configurations and ordering) and $\ZSSP$?}
\end{quote}
A tight construction (instance achieving the gap) and a matching upper bound proof
are both needed.
\end{openproblem}

\begin{openproblem}
\textbf{TODO 2: Effect of sub-optimal groupings on the gap.}

\medskip
The Theorem~\ref{thm:unbounded} construction uses JGP-optimal groupings.
A sub-optimal grouping with $K > K^*$ groups has more slack per group,
potentially enabling lookahead configurations with smaller transitions.
It is conjectured that some sub-optimal groupings yield $H < $ the best
JGP-optimal result. Proving (or disproving) this requires:
\begin{itemize}
\item A formal characterisation of which sub-optimal groupings are beneficial.
\item A bound on how much extra slack is needed to close the gap.
\end{itemize}
See \texttt{06\_grouping\_selection.tex} for related directions.
\end{openproblem}
