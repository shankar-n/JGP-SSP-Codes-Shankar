\section{Gap and Approximation Ratio (Part V)}

\begin{op}
\textbf{OP\,1 (Tight ratio for $b=3$; conjecture).} Is $H/\ZSSP \le 4/3$ for every
$b=3$ instance, with equality only at the $6$-job ring?
\emph{Status:} open. Strong evidence --- no counterexample in $10691$ exhaustively
enumerated $b=3$ edge instances, nor in the copy-paste or ring families (Part~V). The
earlier route via $\sum_k R_k \le H/4$ is \emph{false} (counterexample
$\{a,b,c\},\{a,b,d\},\{c,d,e\}$) and has been retracted.
\end{op}

\begin{op}
\textbf{OP\,2 (Extremal ratio for general $b$).} What is $\sup_I H/\ZSSP$ as a function
of $b$? \emph{Status:} open. Verified data: $b=3$ attains $4/3$; the $b=4$ two-tool
family attains only $1$, while a $b=4$ three-tool sliding ring attains $6/5$. The
dependence on $b$ (and on tool-set sizes) is not characterised.
\end{op}

\begin{op}
\textbf{OP\,3 (Structural characterisation of zero gap).} Give a necessary-and-sufficient
structural condition for $H=\ZSSP$. \emph{Status:} open. Two \emph{sufficient} conditions
are proved (the heuristic meets either lower bound $\Kstar-1$ or $|U|-b$; Part~V); and
empirically every positive-gap $b=3$ edge instance has $\Kstar=3$. A full characterisation
(e.g.\ via boundary tool-overlap / hypergraph structure) is missing.
\end{op}

\section{Grouping Selection (Parts V--VI)}

\begin{op}
\textbf{OP\,4 (Joint grouping--sequencing).} The JGP+GSP heuristic fixes a JGP-\emph{optimal}
grouping ($K=\Kstar$). It is \emph{proved} (Part~V) that a sub-optimal grouping ($K>\Kstar$)
can strictly beat it --- on the $6$-ring a $K=4$ grouping attains $\ZSSP=3<4=H$. The open
problem is to characterise, or efficiently optimise over, the joint choice of grouping and
sequencing; the relevant objective is the joint cost, not $\Kstar$.
\end{op}

\begin{op}
\textbf{OP\,5 (Beyond one-directional lookahead).} The MWHP ordering result (Part~VI) is
exact only for the one-directional lookahead policy and coincides with the optimum at zero
slack; with positive slack, bidirectional padding can be strictly cheaper (the
sliding-window family has true cost $|U|-b=(K{-}1)(r{-}\mathrm{ov})-(b{-}r)$, below the MWHP
value $(K{-}1)(r{-}\mathrm{ov})$). A polynomial exact GSP for positive-slack groupings, or a
proof of NP-hardness, is open.
\end{op}

\section{Exact Formulations (Parts II, IV)}

\begin{op}
\textbf{OP\,6 (The exchange argument; central to Part~IV).} Does every uniform-SSP instance
admit a \emph{cluster-simple} optimal configuration path (each ordered cluster pair traversed
at most once)? If yes, the cluster-level MTZ formulation is exact; if no, it is only a
restriction (heuristic upper bound). \emph{Status:} open. Proved for pairwise-incompatible
instances and the $6$-ring (Part~IV).
\end{op}

\begin{op}
\textbf{OP\,7 (LP strength and net solver benefit of MTZ).} The cluster-MTZ relaxation is no
tighter than SEC and is weakened further by aggregation (Part~IV). Quantify the LP gap on SSP
instances (with and without Desrochers--Laporte lifting), and measure whether removing row
generation yields a net branch-and-price speed-up versus the SEC and SSPMF formulations on the
Catanzaro Tabela1C set. \emph{Status:} open, computational.
\end{op}

\begin{op}
\textbf{OP\,8 (Fractional Benders cuts).} The branch-and-Benders solver currently adds
optimality cuts only at integer master solutions (KTNS subproblem). Deriving valid cuts at
\emph{fractional} master solutions from a relaxation of the KTNS dual --- enabling user cuts
that tighten the bound before branching --- is open.
\end{op}

\section{Cost-Function Variants (Part VII)}

\begin{op}
\textbf{OP\,9 (Tight collapse condition).} The conditions (C1)+(C2)+(C3) are sufficient for
collapse of an SSP variant to the JGP; the necessity analysis (Part~VII) shows (C3) necessary
and (C1),(C2) dispensable. A single necessary-and-sufficient characterisation of collapse,
beyond the reduction to the threshold $\rho_c(I)$, is open.
\end{op}

\begin{op}
\textbf{OP\,10 (Setup-time regime in practice).} For the magazine/tool setup-time ratio
$\rho$, the TD-SSP approximation ratio $R(\rho)$ decreases from $\le 4/3$ at $\rho=0$ to $1$
as $\rho\to\infty$, with ring threshold $\rho_c=1$ (Part~VII). Whether the manufacturing
regime $\rho\in[3,60]$ (Privault--Finke) makes JGP+GSP exactly optimal on broader instance
classes (not just ring-like) is open.
\end{op}
